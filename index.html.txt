    let currentCategory = '';
    let currentQ = 0;
    let totalScore = 0;

    function selectCategory(cat) {
        currentCategory = cat;
        document.getElementById('screen-start').classList.add('hidden');
        document.getElementById('screen-quiz').classList.remove('hidden');
        showQuestion();
    }

    function showQuestion() {
        const qData = questions[currentQ];
        document.getElementById('question-text').innerText = qData.q;
        
        // 프로그레스 바 업데이트
        const percent = ((currentQ + 1) / questions.length) * 100;
        document.getElementById('progress').style.width = percent + '%';

        const answersDiv = document.getElementById('answers');
        answersDiv.innerHTML = '';
        
        qData.options.forEach(opt => {
            const btn = document.createElement('button');
            btn.innerText = opt.text;
            btn.onclick = () => selectAnswer(opt.score);
            answersDiv.appendChild(btn);
        });
    }

    function selectAnswer(score) {
        totalScore += score;
        currentQ++;
        if (currentQ < questions.length) {
            showQuestion();
        } else {
            showResult();
        }
    }

    function showResult() {
        document.getElementById('screen-quiz').classList.add('hidden');
        document.getElementById('screen-result').classList.remove('hidden');

        // 점수에 따라 아이돌 리스트 중 1명 매칭
        const list = idolData[currentCategory];
        const resultIndex = totalScore % list.length;
        const result = list[resultIndex];

        document.getElementById('res-badge').innerText = result.tag;
        document.getElementById('res-name').innerText = result.name;
        document.getElementById('res-desc').innerText = result.desc;
    }

    function resetTest() {
        currentQ = 0;
        totalScore = 0;
        document.getElementById('screen-result').classList.add('hidden');
        document.getElementById('screen-start').classList.remove('hidden');
    }

        navigator.clipboard.writeText(window.location.href);
